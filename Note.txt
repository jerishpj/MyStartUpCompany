 Azure Infrastructure Setup
---------------------------------
Option A: Using Azure Portal (Manual Setup)

1.	Create Resource Group:

az group create --name rg-mystartupcompany-prod --location eastus

2.	Create Azure SQL Server & Database:

az sql server create \
  --name sql-mystartupcompany-prod \
  --resource-group rg-mystartupcompany-prod \
  --location eastus \
  --admin-user sqladmin \
  --admin-password 'YourStrongPassword123!'

az sql db create \
  --resource-group rg-mystartupcompany-prod \
  --server sql-mystartupcompany-prod \
  --name MyStartUpCompanyDb \
     --service-objective S0
   
   # Configure firewall to allow Azure services
   az sql server firewall-rule create \
     --resource-group rg-mystartupcompany-prod \
     --server sql-mystartupcompany-prod \
     --name AllowAzureServices \
     --start-ip-address 0.0.0.0 \
     --end-ip-address 0.0.0.0

3.	Create Azure Container Registry:


az acr create \
     --resource-group rg-mystartupcompany-prod \
     --name acrmystartupcompany \
     --sku Basic \
     --admin-enabled true

4.	Create Azure App Service Plan:


az appservice plan create \
     --name plan-mystartupcompany-prod \
     --resource-group rg-mystartupcompany-prod \
     --is-linux \
     --sku P1V2


5.	Create Azure App Service (Web App for Containers):

az webapp create \
     --resource-group rg-mystartupcompany-prod \
     --plan plan-mystartupcompany-prod \
     --name app-mystartupcompany-api-prod \
     --deployment-container-image-name acrmystartupcompany.azurecr.io/mystartupcompany-api:latest
   
   # Enable continuous deployment
   az webapp deployment container config \
     --name app-mystartupcompany-api-prod \
     --resource-group rg-mystartupcompany-prod \
     --enable-cd true

Option B: Using Bicep (Infrastructure as Code - Recommended)



@description('Environment name (e.g., prod, staging)')
param environmentName string = 'prod'

@description('Location for all resources')
param location string = resourceGroup().location

@description('SQL Server administrator login')
@secure()
param sqlAdminUsername string

@description('SQL Server administrator password')
@secure()
param sqlAdminPassword string

var appName = 'mystartupcompany'
var sqlServerName = 'sql-${appName}-${environmentName}'
var sqlDatabaseName = 'MyStartUpCompanyDb'
var acrName = 'acr${appName}${environmentName}'
var appServicePlanName = 'plan-${appName}-${environmentName}'
var webAppName = 'app-${appName}-api-${environmentName}'

// Azure Container Registry
resource containerRegistry 'Microsoft.ContainerRegistry/registries@2023-01-01-preview' = {
  name: acrName
  location: location
  sku: {
    name: 'Basic'
  }
  properties: {
    adminUserEnabled: true
  }
}

// SQL Server
resource sqlServer 'Microsoft.Sql/servers@2022-05-01-preview' = {
  name: sqlServerName
  location: location
  properties: {
    administratorLogin: sqlAdminUsername
    administratorLoginPassword: sqlAdminPassword
    version: '12.0'
  }
}

// SQL Database
resource sqlDatabase 'Microsoft.Sql/servers/databases@2022-05-01-preview' = {
  parent: sqlServer
  name: sqlDatabaseName
  location: location
  sku: {
    name: 'S0'
    tier: 'Standard'
  }
}

// SQL Firewall Rule - Allow Azure Services
resource sqlFirewallRule 'Microsoft.Sql/servers/firewallRules@2022-05-01-preview' = {
  parent: sqlServer
  name: 'AllowAzureServices'
  properties: {
    startIpAddress: '0.0.0.0'
    endIpAddress: '0.0.0.0'
  }
}

// App Service Plan
resource appServicePlan 'Microsoft.Web/serverfarms@2022-09-01' = {
  name: appServicePlanName
  location: location
  kind: 'linux'
  properties: {
    reserved: true
  }
  sku: {
    name: 'P1V2'
    tier: 'PremiumV2'
  }
}

// App Service (Web App for Containers)
resource webApp 'Microsoft.Web/sites@2022-09-01' = {
  name: webAppName
  location: location
  properties: {
    serverFarmId: appServicePlan.id
    siteConfig: {
      linuxFxVersion: 'DOCKER|${containerRegistry.properties.loginServer}/mystartupcompany-api:latest'
      appSettings: [
        {
          name: 'DOCKER_REGISTRY_SERVER_URL'
          value: 'https://${containerRegistry.properties.loginServer}'
        }
        {
          name: 'DOCKER_REGISTRY_SERVER_USERNAME'
          value: containerRegistry.listCredentials().username
        }
        {
          name: 'DOCKER_REGISTRY_SERVER_PASSWORD'
          value: containerRegistry.listCredentials().passwords[0].value
        }
        {
          name: 'WEBSITES_ENABLE_APP_SERVICE_STORAGE'
          value: 'false'
        }
        {
          name: 'ASPNETCORE_ENVIRONMENT'
          value: 'Production'
        }
      ]
      connectionStrings: [
        {
          name: 'DefaultConnection'
          connectionString: 'Server=tcp:${sqlServer.properties.fullyQualifiedDomainName},1433;Database=${sqlDatabaseName};User ID=${sqlAdminUsername};Password=${sqlAdminPassword};Encrypt=True;TrustServerCertificate=False;Connection Timeout=30;'
          type: 'SQLAzure'
        }
      ]
    }
  }
}

output webAppUrl string = webApp.properties.defaultHostName
output acrLoginServer string = containerRegistry.properties.loginServer

Azure DevOps Setup Steps
==============================

A. Create Service Connections
1.	Go to Project Settings → Service connections
2.	Create Azure Resource Manager connection:
•	Name: azure-mystartupcompany-prod
•	Select your subscription
•	Resource group: rg-mystartupcompany-prod
3.	Create Docker Registry connection:
•	Registry type: Azure Container Registry
•	Azure subscription: (select yours)
•	Azure container registry: acrmystartupcompany
B. Create Variable Groups
1.	Go to Pipelines → Library → + Variable group
2.	Create mystartupcompany-prod with:
•	sqlServerName: sql-mystartupcompany-prod
•	sqlDatabaseName: MyStartUpCompanyDb
•	sqlAdminUsername: sqladmin
•	sqlAdminPassword: YourPassword (mark as secret 🔒)
C. Create Environments
1.	Go to Pipelines → Environments → New environment
2.	Create environments: production and staging
3.	Add approvals for production environment
